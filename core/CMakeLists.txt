set(DC_WARN -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare)
set(DC_OPT)
if(DC_NATIVE AND NOT CMAKE_CROSSCOMPILING)
  include(CheckCXXCompilerFlag)
  check_cxx_compiler_flag("-march=native" DC_HAS_MARCH_NATIVE)
  if(DC_HAS_MARCH_NATIVE)
    list(APPEND DC_OPT -march=native)
  else()
    check_cxx_compiler_flag("-mcpu=native" DC_HAS_MCPU_NATIVE)
    if(DC_HAS_MCPU_NATIVE)
      list(APPEND DC_OPT -mcpu=native)
    endif()
  endif()
endif()
if(DC_SANITIZE)
  list(APPEND DC_OPT -fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

file(GLOB_RECURSE DC_CORE_SRC CONFIGURE_DEPENDS
  ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cc)
list(FILTER DC_CORE_SRC EXCLUDE REGEX ".*/src/tools/.*")

add_library(dccore STATIC ${DC_CORE_SRC})
target_include_directories(dccore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_options(dccore PRIVATE ${DC_WARN} ${DC_OPT})
target_link_libraries(dccore PUBLIC
  h3 utf8proc nlohmann_json::nlohmann_json httplib::httplib Threads::Threads)

function(dc_tool name src)
  add_executable(${name} ${src})
  target_compile_options(${name} PRIVATE ${DC_WARN} ${DC_OPT})
  target_link_libraries(${name} PRIVATE dccore)
endfunction()

# Services and offline tools.
dc_tool(dishcovery-shard   src/tools/shard_main.cc)
dc_tool(dishcovery-indexer src/tools/indexer_main.cc)
dc_tool(dishcovery-suggest src/tools/suggest_main.cc)
dc_tool(dishcovery-geostat src/tools/geostat_main.cc)

if(DC_BUILD_BENCH)
  file(GLOB DC_BENCH_SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/bench/*.cc)
  foreach(src ${DC_BENCH_SRC})
    get_filename_component(bname ${src} NAME_WE)
    dc_tool(${bname} ${src})
  endforeach()
endif()

if(DC_BUILD_TESTS)
  file(GLOB DC_TEST_SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cc)
  add_executable(dishcovery_tests ${DC_TEST_SRC})
  target_include_directories(dishcovery_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
  target_compile_options(dishcovery_tests PRIVATE ${DC_WARN} ${DC_OPT})
  target_link_libraries(dishcovery_tests PRIVATE dccore)
  enable_testing()
  add_test(NAME dishcovery_tests COMMAND dishcovery_tests)
endif()
